= $state 用在 class 里的验证

本篇验证 Svelte 5 官方文档 https://svelte.dev/docs/svelte/$state#Classes 中的写法：
*class 实例本身不会被 proxy 代理，但可以把 `$state` 用在 class 字段声明，或构造函数里对属性的首次赋值上。*

环境：Svelte `^5.56.1`，`vite.config.js` 已强制开启 `runes: true`，Vitest（server + client 两个项目）。

== 1. 官方示例（要验证的写法）

[source,js]
----
class Todo {
	done = $state(false);          // 字段声明处使用 $state

	constructor(text) {
		this.text = $state(text);   // 构造函数中首次赋值处使用 $state
	}

	reset() {
		this.text = '';
		this.done = false;
	}
}
----

== 3. 响应式追踪辅助

`$effect` 在 node/SSR 编译产物里是空实现，所以响应式断言必须跑在 client（浏览器）项目。`tracker.svelte.js` 在 `$effect.root` 里订阅一个 getter：

[source,js]
----
import { flushSync } from 'svelte';

export function createTracker(getter) {
	let value = undefined;
	let runs = 0;

	const destroy = $effect.root(() => {
		$effect(() => {
			value = getter();
			runs += 1;
		});
	});

	flushSync();

	return {
		get value() { return value; },
		get runs() { return runs; },
		run(mutate) { flushSync(mutate); },
		destroy
	};
}
----

== 4. 验证结论

=== 4.1 两种写法都生效 ✅

[cols="2,3,1"]
|===
| 写法 | 行为 | 结论

| `done = $state(false)`（字段声明）
| 改 `todo.done` 时 effect 重跑（runs 1 → 2）
| ✅ 响应式生效

| `this.text = $state(text)`（构造函数首次赋值）
| 改 `todo.text` 时 effect 重跑
| ✅ 响应式生效

| `#done = $state(false)`（private 字段）
| 编译为 `$.state(false)` + `$.get/$.set`，effect 重跑
| ✅ 也生效

| 调用 `reset()` 后两个字段仍响应式
| 两个 effect 在重置后各自再重跑
| ✅ 生效
|===

=== 4.2 实例确实没有被 proxy ✅

给实例后加一个普通属性，值写进去了，但 effect **没有**重跑——这正是文档说的「只对显式包了 `$state` 的字段生效」：

[source,js]
----
/** @type {Todo & { note?: string }} */
const todo = new Todo('learn svelte');
const tracker = createTracker(() => todo.note);

tracker.run(() => {
	todo.note = 'not reactive'; // 写进去了
});

expect(todo.note).toBe('not reactive');
expect(tracker.runs).toBe(1); // effect 没重跑 => 实例不是 proxy
----

=== 4.3 解构会丢失响应式 ✅

[source,js]
----
const { text } = todo;          // 只是快照
const tracker = createTracker(() => text);

tracker.run(() => { todo.text = 'changed'; });

expect(todo.text).toBe('changed');
expect(text).toBe('learn svelte'); // 局部变量不会跟着变
expect(tracker.runs).toBe(1);
----

== 5. 「首次赋值」是编译期强制的

通过 `compileModule(source, { generate: 'client' })` 直接编译源码得到验证：

[cols="2,3"]
|===
| 写法 | 编译结果

| 文档里的两种写法 / private 字段
| 正常编译（无报错）

| 构造函数里先 `this.text = text` 再 `this.text = $state(text)`
| 报错 `Cannot assign to a state field before its declaration`

| 在普通方法里 `this.note = $state('x')`
| 报错 `state_invalid_placement`（消息要求写在构造函数顶层对字段的首次赋值）
|===

== 8. 测试源码附录

以下为验证用文件的完整源码（与项目内 `src/lib/state-class/` 下的真实文件一致），可直接对照。

=== 8.1 `Todo.svelte.js`

[source,js]
----
/* global $state */

/**
 * 验证 https://svelte.dev/docs/svelte/$state#Classes 中的写法。
 *
 * 官方说明：class 实例本身**不会**被 proxy 代理，但可以分别在
 *  1. class 字段声明处（public / private 都可）
 *  2. 构造函数里对某个属性的**首次**赋值处
 * 使用 `$state`，从而让单个属性具备响应式。
 *
 * 注意：文件名必须是 `*.svelte.js`，模块里的 runes 才会被编译。
 */
export class Todo {
	/** 字段声明处使用 $state */
	done = $state(false);

	/**
	 * @param {string} text
	 */
	constructor(text) {
		/** 构造函数中对该属性的首次赋值处使用 $state */
		this.text = $state(text);
	}

	/** 把两个响应式字段一起重置 */
	reset() {
		this.text = '';
		this.done = false;
	}
}
----

=== 8.2 `tracker.svelte.js`

[source,js]
----
/* global $effect */
import { flushSync } from 'svelte';

/**
 * 在 `$effect.root` 里订阅一个 getter，用来观察某个属性**是否真的具备响应式**。
 * runes 只能在 `.svelte` / `.svelte.js` 文件里使用，所以这个辅助函数单独放在 `.svelte.js` 中。
 *
 * @param {() => unknown} getter 读取待观察属性的函数
 */
export function createTracker(getter) {
	/** @type {unknown} */
	let value = undefined;

	/** effect 被执行的次数 */
	let runs = 0;

	const destroy = $effect.root(() => {
		$effect(() => {
			value = getter();
			runs += 1;
		});
	});

	// flush 一次，让首次读取的结果落到 value 上
	flushSync();

	return {
		/** effect 最近一次读到的值 */
		get value() {
			return value;
		},
		/** effect 被执行了多少次 */
		get runs() {
			return runs;
		},
		/**
		 * 修改状态并同步 flush，确保 effect 已经重跑
		 * @param {() => void} mutate
		 */
		run(mutate) {
			flushSync(mutate);
		},
		destroy
	};
}
----

=== 8.3 `Todo.svelte.spec.js`

[source,js]
----
import { describe, expect, it } from 'vitest';
import { Todo } from './Todo.svelte.js';
import { createTracker } from './tracker.svelte.js';

// 注意：这个文件的名字是 *.svelte.spec.js，会跑在 vitest 的 client（浏览器）项目里。
// 原因是 Svelte 在 node/SSR 编译产物中 $effect` 不会执行，只有客户端才能真正验证响应式。
// 同理，Todo.svelte.js / tracker.svelte.js 必须带 .svelte.js 后缀，runes 才会被编译。

describe('Todo：把 $state` 用在 class 里', () => {
	it('字段声明处的 $state(false) 初始值正确', () => {
		const todo = new Todo('learn svelte');

		expect(todo.done).toBe(false);
	});

	it('构造函数首次赋值处的 $state(text) 初始值正确', () => {
		const todo = new Todo('learn svelte');

		expect(todo.text).toBe('learn svelte');
	});

	it('修改 done（字段声明式 $state`）会触发响应式更新', () => {
		const todo = new Todo('learn svelte');
		const tracker = createTracker(() => todo.done);

		expect(tracker.value).toBe(false);
		expect(tracker.runs).toBe(1);

		tracker.run(() => {
			todo.done = true;
		});

		expect(todo.done).toBe(true);
		expect(tracker.value).toBe(true);
		expect(tracker.runs).toBe(2);
	});

	it('修改 text（构造函数式 $state`）会触发响应式更新', () => {
		const todo = new Todo('learn svelte');
		const tracker = createTracker(() => todo.text);

		expect(tracker.value).toBe('learn svelte');

		tracker.run(() => {
			todo.text = 'learn runes';
		});

		expect(todo.text).toBe('learn runes');
		expect(tracker.value).toBe('learn runes');
		expect(tracker.runs).toBe(2);
	});

	it('reset() 同时重置两个字段，并各自触发更新', () => {
		const todo = new Todo('learn svelte');
		const doneTracker = createTracker(() => todo.done);
		const textTracker = createTracker(() => todo.text);

		doneTracker.run(() => {
			todo.done = true;
			todo.text = 'changed';
		});

		expect(doneTracker.value).toBe(true);
		expect(textTracker.value).toBe('changed');
		expect(doneTracker.runs).toBe(2);
		expect(textTracker.runs).toBe(2);

		textTracker.run(() => todo.reset());

		expect(todo.text).toBe('');
		expect(todo.done).toBe(false);
		expect(textTracker.value).toBe('');
		expect(doneTracker.value).toBe(false);
		expect(textTracker.runs).toBe(3);
		expect(doneTracker.runs).toBe(3);
	});

	it('实例本身没有被 proxy：后加的普通属性不具备响应式', () => {
		/** @type {Todo & { note?: string }} */
		const todo = new Todo('learn svelte');
		const tracker = createTracker(() => todo.note);

		expect(tracker.runs).toBe(1);

		tracker.run(() => {
			todo.note = 'not reactive';
		});

		// 值确实写进去了，但 effect 没有重跑 —— 说明实例不是 proxy
		expect(todo.note).toBe('not reactive');
		expect(tracker.runs).toBe(1);
	});
});
----

=== 8.4 `edge-cases.svelte.js`

[source,js]
----
/* global $state */

/**
 * 文档里说 `$state` 可以用在 class 字段上，「无论是 public 还是 private」。
 * 这里验证 private（`#` 私有字段）的写法，并通过 getter / setter 暴露。
 */
export class PrivateTodo {
	#done = $state(false);

	/**
	 * @param {string} text
	 */
	constructor(text) {
		this.text = $state(text);
	}

	get done() {
		return this.#done;
	}

	/**
	 * @param {boolean} value
	 */
	set done(value) {
		this.#done = value;
	}

	reset() {
		this.text = '';
		this.done = false;
	}
}
----

=== 8.5 `edge-cases.svelte.spec.js`

[source,js]
----
import { describe, expect, it } from 'vitest';
import { PrivateTodo } from './edge-cases.svelte.js';
import { createTracker } from './tracker.svelte.js';

// 跑在 vitest 的 client（浏览器）项目里：只有客户端编译产物中 $effect` 才会真正执行。

describe('边界情况', () => {
	it('private 字段 #done = $state(false) 同样具备响应式', () => {
		const todo = new PrivateTodo('learn svelte');
		const tracker = createTracker(() => todo.done);

		expect(todo.done).toBe(false);
		expect(tracker.runs).toBe(1);

		tracker.run(() => {
			todo.done = true;
		});

		expect(todo.done).toBe(true);
		expect(tracker.value).toBe(true);
		expect(tracker.runs).toBe(2);
	});

	it('解构出来的值只是一份快照，不再具备响应式', () => {
		const todo = new PrivateTodo('learn svelte');
		const { text } = todo;
		const tracker = createTracker(() => text);

		expect(text).toBe('learn svelte');

		tracker.run(() => {
			todo.text = 'changed';
		});

		// 实例上的字段变了，但之前解构出来的局部变量不会跟着变
		expect(todo.text).toBe('changed');
		expect(text).toBe('learn svelte');
		expect(tracker.runs).toBe(1);
	});
});

=== 8.6 `edge-cases.spec.js`

[source,js]
----
import { compileModule } from 'svelte/compiler';
import { describe, expect, it } from 'vitest';

// 这个用例跑在 server（node）项目里：只做编译期校验，不需要 $effect`，也就不需要浏览器。

/**
 * 按 client 模式编译一段源码，返回报错信息；没有报错则返回 null。
 * @param {string} source
 */
function compileError(source) {
	try {
		compileModule(source, { generate: 'client', filename: 'case.svelte.js' });
		return null;
	} catch (error) {
		return /** @type {Error} */ (error).message;
	}
}

describe('$state` 在 class 里的合法 / 非法位置', () => {
	it('文档里的两种写法都能正常编译', () => {
		const message = compileError(`
			export class Todo {
				done = $state(false);

				constructor(text) {
					this.text = $state(text);
				}
			}
		`);

		expect(message).toBeNull();
	});

	it('private 字段也能正常编译', () => {
		const message = compileError(`
			export class Todo {
				#done = $state(false);
			}
		`);

		expect(message).toBeNull();
	});

	it('构造函数里「非首次」赋值会编译报错', () => {
		const message = compileError(`
			export class Todo {
				constructor(text) {
					this.text = text;
					this.text = $state(text);
				}
			}
		`);

		expect(message).toContain('Cannot assign to a state field before its declaration');
	});

	it('在普通方法里用 $state` 赋值会编译报错', () => {
		const message = compileError(`
			export class Todo {
				addNote() {
					this.note = $state('x');
				}
			}
		`);

		expect(message).toContain(
			'the first assignment to a class field at the top level of the constructor'
		);
	});
});

== 9. 小结

* 文档里的 `Todo` 类写法**完全正确**。
* `$state` 可放在：class 字段声明（public 或 private）、构造函数里对属性的**首次**赋值。
* 实例不被 proxy：只有显式用 `$state` 包装的字段才有响应式；后加属性、解构出来的变量都不具备响应式。
* 「首次赋值」由编译器强制，写错位置会直接报编译错误。
